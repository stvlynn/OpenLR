cmake_minimum_required(VERSION 3.20)
project(OpenLRJxlColor LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(OPENLR_THREADS "Build the pthread encoder variant" OFF)
option(OPENLR_SIMD "Enable WASM SIMD throughout libjxl and the wrapper" OFF)
option(OPENLR_JXL_EXPERIMENTS "Expose validation-only compression mode control" OFF)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Configure through emcmake; this target produces WebAssembly")
endif()
if(NOT OPENLR_JXL_SOURCE)
  message(FATAL_ERROR "Pass -DOPENLR_JXL_SOURCE=<pinned libjxl source directory>")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
foreach(feature TOOLS MANPAGES BENCHMARK EXAMPLES FUZZERS DEVTOOLS JNI SJPEG
    OPENEXR VIEWERS TCMALLOC PLUGINS)
  set(JPEGXL_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(JPEGXL_ENABLE_SKCMS ON CACHE BOOL "" FORCE)
set(JPEGXL_BUNDLE_SKCMS ON CACHE BOOL "" FORCE)
set(JPEGXL_VERSION "9f544641ec83f6abd9da598bdd08178ee8a003e0" CACHE STRING "" FORCE)
set(HWY_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(HWY_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
add_compile_options(-O3 -Wno-deprecated-declarations)
if(OPENLR_THREADS)
  add_compile_options(-pthread)
  add_link_options(-pthread "-sPTHREAD_POOL_SIZE=navigator.hardwareConcurrency")
endif()
if(OPENLR_SIMD)
  add_compile_options(-msimd128)
  add_link_options(-msimd128)
endif()
add_subdirectory("${OPENLR_JXL_SOURCE}" libjxl EXCLUDE_FROM_ALL)
set(bindings jxl_color_enc)
if(NOT OPENLR_THREADS AND NOT OPENLR_SIMD)
  list(APPEND bindings jxl_inspect jxl_color_dec jxl_input_fixture)
endif()
foreach(binding ${bindings})
  add_executable(${binding} "${binding}.cpp")
  set_target_properties(${binding} PROPERTIES SUFFIX ".js")
  target_link_libraries(${binding} PRIVATE jxl-static)
  if(binding STREQUAL "jxl_color_dec")
    target_include_directories(${binding} PRIVATE $<TARGET_PROPERTY:skcms,INCLUDE_DIRECTORIES>)
    # Reuse libjxl's bundled skcms objects, whose exported C names are prefixed.
    target_compile_definitions(${binding} PRIVATE JPEGXL_BUNDLE_SKCMS=1)
    target_compile_options(${binding} PRIVATE -include "${OPENLR_JXL_SOURCE}/lib/jxl/enc_jxl_skcms.h")
  endif()
  if(OPENLR_THREADS)
    target_link_libraries(${binding} PRIVATE jxl_threads-static)
  endif()
  target_link_options(${binding} PRIVATE --bind -O3
    "-sENVIRONMENT=web,worker,node" "-sFILESYSTEM=0" "-sALLOW_MEMORY_GROWTH=1"
    "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sDYNAMIC_EXECUTION=0")
endforeach()
if(OPENLR_THREADS AND OPENLR_SIMD)
  set_target_properties(jxl_color_enc PROPERTIES OUTPUT_NAME "jxl_color_enc_mt_simd")
elseif(OPENLR_THREADS)
  set_target_properties(jxl_color_enc PROPERTIES OUTPUT_NAME "jxl_color_enc_mt")
endif()
if(OPENLR_JXL_EXPERIMENTS)
  target_compile_definitions(jxl_color_enc PRIVATE OPENLR_JXL_EXPERIMENTS=1)
endif()
