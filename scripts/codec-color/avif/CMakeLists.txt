# OpenLR build integration; upstream codec source licenses remain unchanged.
cmake_minimum_required(VERSION 3.20)
project(openlr_avif_color LANGUAGES C CXX)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Configure with emcmake and Emscripten 3.1.64")
endif()
if(NOT DEFINED CODEC_SOURCE_DIR)
  message(FATAL_ERROR "CODEC_SOURCE_DIR must contain libavif, libaom, libwebp")
endif()
option(OPENLR_THREADS "Build the threaded encoder" OFF)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
find_package(Threads REQUIRED)
if(OPENLR_THREADS)
  add_compile_options(-pthread)
  add_link_options(-pthread)
endif()
set(AOM_TARGET_CPU generic CACHE STRING "" FORCE)
foreach(flag ENABLE_CCACHE ENABLE_DOCS ENABLE_TESTS ENABLE_EXAMPLES ENABLE_TOOLS CONFIG_INSPECTION CONFIG_RUNTIME_CPU_DETECT CONFIG_WEBM_IO)
  set(${flag} 0 CACHE BOOL "" FORCE)
endforeach()
if(OPENLR_THREADS)
  set(CONFIG_MULTITHREAD 1 CACHE STRING "" FORCE)
else()
  set(CONFIG_MULTITHREAD 0 CACHE STRING "" FORCE)
endif()
set(CONFIG_AV1_HIGHBITDEPTH 1 CACHE BOOL "" FORCE)
set(CONFIG_ACCOUNTING 1 CACHE BOOL "" FORCE)
add_subdirectory("${CODEC_SOURCE_DIR}/libaom" libaom EXCLUDE_FROM_ALL)

# Build only sharpyuv, retaining the existing jSquash option without libwebp tools.
foreach(flag WEBP_BUILD_ANIM_UTILS WEBP_BUILD_CWEBP WEBP_BUILD_DWEBP WEBP_BUILD_GIF2WEBP WEBP_BUILD_IMG2WEBP WEBP_BUILD_VWEBP WEBP_BUILD_WEBPINFO WEBP_BUILD_LIBWEBPMUX WEBP_BUILD_WEBPMUX WEBP_BUILD_EXTRAS)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(WEBP_USE_THREAD ${OPENLR_THREADS} CACHE BOOL "" FORCE)
add_subdirectory("${CODEC_SOURCE_DIR}/libwebp" libwebp EXCLUDE_FROM_ALL)
set(libsharpyuv_FOUND TRUE)
set(LIBSHARPYUV_LIBRARY sharpyuv)
set(LIBSHARPYUV_INCLUDE_DIR "${CODEC_SOURCE_DIR}/libwebp")

set(AVIF_CODEC_AOM ON CACHE BOOL "" FORCE)
set(AVIF_CODEC_AOM_ENCODE ON CACHE BOOL "" FORCE)
set(AVIF_CODEC_AOM_DECODE OFF CACHE BOOL "" FORCE)
set(AVIF_ENABLE_WERROR OFF CACHE BOOL "" FORCE)
set(AVIF_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(AVIF_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(AOM_INCLUDE_DIR "${CODEC_SOURCE_DIR}/libaom")
set(AOM_LIBRARIES aom)
add_subdirectory("${CODEC_SOURCE_DIR}/libavif" libavif EXCLUDE_FROM_ALL)
target_include_directories(avif PRIVATE "${CODEC_SOURCE_DIR}/libaom")

if(NOT OPENLR_THREADS)
  # Reuse upstream's actual source list and build properties. Separate codec
  # tables let wasm-ld discard the unused libaom encoder/decoder archive objects.
  get_target_property(avif_sources avif SOURCES)
  set(decode_sources)
  foreach(source IN LISTS avif_sources)
    if(IS_ABSOLUTE "${source}")
      list(APPEND decode_sources "${source}")
    else()
      list(APPEND decode_sources "${CODEC_SOURCE_DIR}/libavif/${source}")
    endif()
  endforeach()
  add_library(avif_decode STATIC ${decode_sources})
  foreach(property INCLUDE_DIRECTORIES INTERFACE_INCLUDE_DIRECTORIES COMPILE_OPTIONS LINK_LIBRARIES)
    get_target_property(value avif ${property})
    if(value)
      set_property(TARGET avif_decode PROPERTY ${property} "${value}")
    endif()
  endforeach()
  get_target_property(definitions avif COMPILE_DEFINITIONS)
  list(FILTER definitions EXCLUDE REGEX "^AVIF_CODEC_AOM_(ENCODE|DECODE)=")
  list(APPEND definitions AVIF_CODEC_AOM_DECODE=1)
  target_compile_definitions(avif_decode PRIVATE ${definitions})
  set_target_properties(avif_decode PROPERTIES C_VISIBILITY_PRESET hidden)
endif()

function(binding name source library)
  add_executable(${name} "${source}")
  target_link_libraries(${name} PRIVATE ${library} aom sharpyuv)
  target_compile_options(${name} PRIVATE -O3)
  target_link_options(${name} PRIVATE -O3 --bind
    --pre-js "${CMAKE_CURRENT_SOURCE_DIR}/pre.js"
    -sENVIRONMENT=web,worker,node -sEXPORT_ES6=1 -sMODULARIZE=1
    -sDYNAMIC_EXECUTION=0 -sALLOW_MEMORY_GROWTH=1 -sFILESYSTEM=0
    -sSTACK_SIZE=5242880 -sINITIAL_MEMORY=16777216)
  set_target_properties(${name} PROPERTIES SUFFIX ".js")
endfunction()
if(OPENLR_THREADS)
  binding(avif_enc_mt avif_enc.cpp avif)
  # Match the published encoder's ready pool: synchronous libaom joins cannot
  # wait for fresh browser Workers to initialize while the JS thread is blocked.
  target_link_options(avif_enc_mt PRIVATE -sPTHREAD_POOL_SIZE=navigator.hardwareConcurrency)
else()
  binding(avif_enc avif_enc.cpp avif)
  binding(avif_dec avif_dec.cpp avif_decode)
endif()
